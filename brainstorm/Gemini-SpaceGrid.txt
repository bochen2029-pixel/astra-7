#define SECTOR_SIZE 1000000.0 // 1,000 km per Sector Voxel
#define RENDER_SHELL 100000.0 // 100 km Max UE5 Render Radius

// ---------------------------------------------------------

struct alignas(32) AstraCoord {
    int64_t sx, sy, sz; // Macro-Voxel Grid
    double lx, ly, lz;  // Micro-Voxel Offset [-500000.0 to 500000.0]

    __host__ __device__ inline void Renormalize() {
        int64_t dx = round(lx / SECTOR_SIZE);
        int64_t dy = round(ly / SECTOR_SIZE);
        int64_t dz = round(lz / SECTOR_SIZE);

        if(dx != 0 || dy != 0 || dz != 0) {
            sx += dx; lx -= dx * SECTOR_SIZE;
            sy += dy; ly -= dy * SECTOR_SIZE;
            sz += dz; lz -= dz * SECTOR_SIZE;
        }
    }

    // Calculates the absolute distance vector in meters
    __host__ __device__ inline void GetRelativeVector(const AstraCoord& target, double& dx, double& dy, double& dz) const {
        dx = (target.sx - sx) * SECTOR_SIZE + (target.lx - lx);
        dy = (target.sy - sy) * SECTOR_SIZE + (target.ly - ly);
        dz = (target.sz - sz) * SECTOR_SIZE + (target.lz - lz);
    }
};

__device__ double solve_kepler(double M, double e) {
    double E = M;
    #pragma unroll
    for (int iter = 0; iter < 5; ++iter) {
        E = E - (E - e * sin(E) - M) / (1.0 - e * cos(E));
    }
    return E;
}
